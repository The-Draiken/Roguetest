// LEVEL 1+ - Universal Passives
// =====================================

IF
    EnteredCombat(_Char, _Combat)
AND
    GetLevel(_Char, _Level)
AND
    _Level >= 1
AND
    NOT DB_CM_ProcessedL1(_Char)
THEN

// BARBARIAN
IF
    EnteredCombat(_Char, _Combat)
AND
    GetLevel(_Char, _Level)
AND
    _Level >= 6
AND
    IsTagged(_Char, (TAG)BARBARIAN_02913f9a-f696-40cf-acdf-32032afab32c, 1)
AND
    NOT DB_CM_ProcessedL6(_Char)
AND
    Random(3, _RandNum)
AND
    IsPlayer((CHARACTER)_Char,0)
AND
	IsPartyMember((CHARACTER)_Char,0,0)
THEN

    DB_CM_ProcessedL6(_Char);

    DB_CM_BarbRoll(_Char, 0)
